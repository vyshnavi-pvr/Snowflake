SQLAI_SUMMARIZE_AGG — Summarize across multiple rows[38]
-- AI_SUMMARIZE_AGG: Aggregate-level summarization across multiple rows.
-- Unlike SUMMARIZE, this works across rows and handles data larger than context windows.
WITH feedback AS (
            SELECT 'The onboarding was smooth and intuitive.' AS comment
  UNION ALL SELECT 'Documentation could be better, hard to find advanced topics.'
  UNION ALL SELECT 'Love the SQL interface — feels natural for our team.'
  UNION ALL SELECT 'Performance is great, but pricing is confusing.'
  UNION ALL SELECT 'Support response times are excellent.'
)
SELECT AI_SUMMARIZE_AGG(comment) AS aggregated_summary
FROM feedback;
